function(add_lab name)
    add_executable(${name} ${ARGN})
    configure_example(${name})
    set_target_properties(${name} PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${PROJECT_BINARY_DIR}/labs")
    if(BUILD_TESTING)
        add_test(NAME ${name}.correctness COMMAND ${name})
        set_tests_properties(${name}.correctness PROPERTIES TIMEOUT 30)
    endif()
endfunction()

add_lab(lab_runtime_dispatch runtime_dispatch/dispatch.c runtime_dispatch/portable.c)
if(HAVE_SSE AND CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
    set(CMAKE_REQUIRED_FLAGS "-msse4.2")
    check_c_source_compiles("#include <nmmintrin.h>
int main(void) { return (int)_mm_crc32_u8(0, 1); }" HAVE_X86_CRC32C)
    unset(CMAKE_REQUIRED_FLAGS)
    if(HAVE_X86_CRC32C)
        add_library(lab_crc_accelerated OBJECT runtime_dispatch/x86.c)
        configure_example(lab_crc_accelerated)
        target_compile_options(lab_crc_accelerated PRIVATE -msse4.2)
        target_compile_definitions(lab_runtime_dispatch PRIVATE LAB_CRC_X86)
    endif()
elseif(HAVE_AARCH64_NEON AND CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
    set(CMAKE_REQUIRED_FLAGS "-march=armv8-a+crc")
    check_c_source_compiles("#include <arm_acle.h>
int main(void) { return (int)__crc32cb(0, 1); }" HAVE_ARM_CRC32C)
    unset(CMAKE_REQUIRED_FLAGS)
    if(HAVE_ARM_CRC32C)
        add_library(lab_crc_accelerated OBJECT runtime_dispatch/arm.c)
        configure_example(lab_crc_accelerated)
        target_compile_options(lab_crc_accelerated PRIVATE -march=armv8-a+crc)
        target_compile_definitions(lab_runtime_dispatch PRIVATE LAB_CRC_ARM)
    endif()
endif()
if(TARGET lab_crc_accelerated)
    target_sources(lab_runtime_dispatch PRIVATE $<TARGET_OBJECTS:lab_crc_accelerated>)
endif()
if(BUILD_TESTING)
    add_test(NAME lab_runtime_dispatch.portable COMMAND lab_runtime_dispatch portable)
    add_test(NAME lab_runtime_dispatch.hardware COMMAND lab_runtime_dispatch hardware)
    add_test(NAME lab_runtime_dispatch.fallback COMMAND lab_runtime_dispatch auto --disable-hardware)
    set_tests_properties(lab_runtime_dispatch.hardware PROPERTIES SKIP_RETURN_CODE 77)
    set_tests_properties(lab_runtime_dispatch.fallback PROPERTIES PASS_REGULAR_EXPRESSION "backend: portable")
endif()

if(MOVING2ARM_BUILD_PYTHON)
    if(CMAKE_CROSSCOMPILING OR CMAKE_C_FLAGS MATCHES "fsanitize")
        message(FATAL_ERROR "The Python lab needs an unsanitized native build directory; leave MOVING2ARM_BUILD_PYTHON=OFF for cross/sanitizer builds.")
    endif()
    if(APPLE AND CMAKE_OSX_ARCHITECTURES)
        list(FIND CMAKE_OSX_ARCHITECTURES "${CMAKE_HOST_SYSTEM_PROCESSOR}" host_arch_index)
        if(host_arch_index LESS 0)
            message(FATAL_ERROR "Build the Python lab for the running interpreter's architecture, in a separate native directory.")
        endif()
    endif()
    find_package(Python3 3.9 REQUIRED COMPONENTS Interpreter Development.Module)
    Python3_add_library(_moving2arm MODULE WITH_SOABI native_dependency/module.c)
    configure_example(_moving2arm)
    target_link_libraries(_moving2arm PRIVATE moving2arm_native)
    if(APPLE)
        set(module_rpath "@loader_path")
    else()
        set(module_rpath "$ORIGIN")
    endif()
    set_target_properties(_moving2arm PROPERTIES
        LIBRARY_OUTPUT_DIRECTORY "${PROJECT_BINARY_DIR}/native_dependency"
        BUILD_WITH_INSTALL_RPATH TRUE INSTALL_RPATH "${module_rpath}")
    add_custom_target(native_wheel
        COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/native_dependency/package_wheel.py"
            --module $<TARGET_FILE:_moving2arm> --library $<TARGET_FILE:moving2arm_native>
            --output "${PROJECT_BINARY_DIR}/wheelhouse"
        DEPENDS _moving2arm moving2arm_native VERBATIM)
    if(BUILD_TESTING)
        add_test(NAME native_dependency.python_and_wheel
            COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/native_dependency/test_native.py"
                --module $<TARGET_FILE:_moving2arm> --library $<TARGET_FILE:moving2arm_native>
                --output "${PROJECT_BINARY_DIR}/wheel-tests")
    endif()
endif()
